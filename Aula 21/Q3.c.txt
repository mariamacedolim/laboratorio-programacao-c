#include <stdio.h>

typedef struct {
    char nome[50];
    float salario;
    int horasExtras;
} Funcionario;

int main(void) {
    FILE *arquivo;
    Funcionario f;

    arquivo = fopen("funcionarios.txt", "a");

    if (arquivo == NULL) {
        printf("Erro ao abrir o arquivo!\n");
        return 1;
    }

    for (int i = 0; i < 5; i++) {
        printf("\nCadastro do funcionario %d\n", i + 1);

        printf("Nome: ");
        scanf(" %49[^\n]", f.nome);

        printf("Salario: ");
        scanf("%f", &f.salario);

        printf("Horas extras: ");
        scanf("%d", &f.horasExtras);

        fprintf(arquivo, "Nome: %s\n", f.nome);
        fprintf(arquivo, "Salario: %.2f\n", f.salario);
        fprintf(arquivo, "Horas extras: %d\n", f.horasExtras);
        fprintf(arquivo, "----------------------\n");
    }

    fclose(arquivo);

    printf("\nFuncionarios cadastrados com sucesso!\n");

    arquivo = fopen("funcionarios.txt", "r");

    if (arquivo == NULL) {
        printf("Erro ao ler o arquivo!\n");
        return 1;
    }

    printf("\n===== FUNCIONARIOS CADASTRADOS =====\n");

    int caractere;

    while ((caractere = fgetc(arquivo)) != EOF) {
        putchar(caractere);
    }

    fclose(arquivo);

    return 0;
}
